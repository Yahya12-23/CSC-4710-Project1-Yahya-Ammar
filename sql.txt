-- SQL used by the site. The ? marks are filled in by mysql2 in Node.js.
-- To try a query in phpMyAdmin, replace each ? with a real value.

-- Database setup
CREATE DATABASE IF NOT EXISTS web_app
  CHARACTER SET utf8mb4 COLLATE utf8mb4_unicode_ci;
USE web_app;
CREATE TABLE IF NOT EXISTS Users (
    username VARCHAR(50) PRIMARY KEY,
    password VARCHAR(255) NOT NULL,
    firstname VARCHAR(50) NOT NULL,
    lastname VARCHAR(50) NOT NULL,
    salary DECIMAL(12,2) NOT NULL,
    age INTEGER NOT NULL,
    registerday DATE NOT NULL,
    registertime DATETIME NOT NULL,
    signintime DATETIME NULL DEFAULT NULL,
    INDEX idx_users_name (firstname, lastname),
    INDEX idx_users_salary (salary),
    INDEX idx_users_registerday (registerday)
);

-- Create a user account
INSERT INTO Users
  (username, password, firstname, lastname, salary, age, registerday, registertime, signintime)
VALUES (?, ?, ?, ?, ?, ?, CURDATE(), NOW(), NULL);

-- Retrieve password hash for login
SELECT username, password FROM Users WHERE username = ?;
-- Update the last sign-in time
UPDATE Users SET signintime = NOW() WHERE username = ?;

-- First name search
SELECT username, firstname, lastname, salary, age, registerday, signintime
FROM Users WHERE firstname LIKE ? ORDER BY username;
-- Last name search
SELECT username, firstname, lastname, salary, age, registerday, signintime
FROM Users WHERE lastname LIKE ? ORDER BY username;
-- First and last name search
SELECT username, firstname, lastname, salary, age, registerday, signintime
FROM Users WHERE firstname LIKE ? AND lastname LIKE ? ORDER BY username;
-- The backend supplies wildcard patterns for name searches.

-- Find by username
SELECT username, firstname, lastname, salary, age, registerday, signintime
FROM Users WHERE username = ?;

-- Salary range search
SELECT username, firstname, lastname, salary, age, registerday, signintime
FROM Users WHERE salary BETWEEN ? AND ? ORDER BY salary, username;

-- Age range search
SELECT username, firstname, lastname, salary, age, registerday, signintime
FROM Users WHERE age BETWEEN ? AND ? ORDER BY age, username;

-- Users who registered after a given user
SELECT username, firstname, lastname, salary, age, registerday, signintime
FROM Users
WHERE registertime > (SELECT registertime FROM Users WHERE username = ?)
ORDER BY registertime, username;

-- Users who never signed in
SELECT username, firstname, lastname, salary, age, registerday, signintime
FROM Users WHERE signintime IS NULL ORDER BY username;

-- Users who registered the same day as a given user (includes that user)
SELECT username, firstname, lastname, salary, age, registerday, signintime
FROM Users
WHERE registerday = (SELECT registerday FROM Users WHERE username = ?)
ORDER BY username;

-- Users who registered today
SELECT username, firstname, lastname, salary, age, registerday, signintime
FROM Users WHERE registerday = CURDATE() ORDER BY username;
